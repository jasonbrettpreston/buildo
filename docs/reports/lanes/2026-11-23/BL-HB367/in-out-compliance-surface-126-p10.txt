### 11a.1 The binding rule (Q-MF3, 2026-11-04)

Operator verbatim: "isnt that the point of the schema - there might be exceptions ie user input fields but overall teh databse fields are the key we also have ads as well. we just need to be aware of this."

- Every displayed field binds to a source. The schema references the field and nothing else: no old engine names, no older-rule text, no code. The CLOSED binding set (`source`):

  | source | what it binds to |
  |---|---|
  | `catalogue` | the Layer 3 output, keyed by catalogue row id (`scripts/seeds/bylaw/layer3-catalogue.json`, schema `layer3-catalogue-v1`; rows gated by `switchboard.json`, schema `switchboard-v1`) |
  | `content` | the label, help-text and legal-copy table (a seed or a DB content table). Labels, help text and legal copy are NOT an exception: they are content fields and bind like any field |
  | `user_input` | an id in the input contract (the form archetype), e.g. neighbour setbacks, the parking-space question |
  | `ad_slot` | an id in the ad contract (§11a.10) |
  | `external_embed` | a declared external embed: map tiles, payment checkout, analytics. Declared, never coded. RULED 2026-11-04 (Q-MF4 a; Spec 135 FE-21) |

- A field with any other source fails the check.
- The set lives in a PROTECTED file (proposed `scripts/surfaces/_schema/binding-sources.json`, path unverified; to be added to `protected-paths.json` at the freeze lift). Adding a source or a member needs a register row and a panel. The binding lock has a RED fixture: a node with an invented field (Spec 128 Rule 20, Spec 135 FE-18).
- Generation: scenario fields are GENERATED from the catalogue and switchboard into the SEED (`scripts/surfaces/_schema/report-v2-seed/` via `seed-report-v2.mjs`), never into the generated `report-v2.decl.json` (its header is re-checked at build, unverified). A row on in the switchboard yields a field set; off yields none; off beats on (the switchboard `$comment`).
- The retired lineage and `_sqm` names are removed from the report schema. Spec 78 needs no dated note for the report (card answer Q-MF3).

### 11a.3 Closed status list and disclosure codes (Honest display)

`status` is one of: `permitted` | `not_permitted:<clause>` | `not_buildable` | `not_evaluated:<reason>` | `not_modelled` | `at_least` (a lower bound; the catalogue row carries the flag).

`coa_required:<clause>` is NOT in the list; it is added by amendment at the McBylaws increment that lands that route (no later-increment work pulled forward).

Disclosure codes are a closed list generated from the catalogue/content (a declared field set), extended only by an amendment to this section plus a lock. A status or code outside the list: the generator fails AND the renderer throws (never skips).

Rule: an `at_least` value renders "at least" beside the number on every target.

### 11a.7 Golden render states: structured render trees

A golden = the pair in Spec 127 §4.3 (API fixture + rendered tree; 126:435-439, 127:129-134). Rendered tree = nodes in declared order:

```
{ "field": "gfa_max_m2", "source": "catalogue", "label": "<from content>", "value": "212.4 m2", "status": "at_least", "confidence": "medium", "provenance": "estimated", "disclosures": ["..."] }
```

`value` is the formatted string; `label` includes any "at least". A screenshot is never a golden.

Render states (minimum): permitted, not_permitted, not_buildable, not_evaluated, not_modelled, at_least, at_least_with_disclosure, one per closed confidence value, variants (flat vs pitched), houseplex 4 and 6, dual FSI, exception lot, basement line, user_entry (§11a.9), no_user_entry, user_entry_outranked (a verified fact outranks a user entry, Q-MF4 a), ad_slot_present, ad_slot_absent (§11a.10), loading, miss, error. +// manifest's own fence, not a second list) that reads a contract (`inputs.answer === 'contract'`) declares an `error` entry in `states[]`
+// and an `errors.answer` that is neither `none` nor `UNRESEARCHED`.
+// LIMITS (stated, not widened): declarations only, and the error state only. Code is not read: the known break at
+// mobile/src/lib/maxbldHelp.ts:19 (a failed Help open is only logged; task-440 RULE-CHECKS.md:102) is not caught here; it is owned by the

+import { execFileSync } from 'node:child_process';
+import { readFileSync } from 'node:fs';
+import { join } from 'node:path';
+import { describe, expect, it } from 'vitest';
+
+const ROOT = process.cwd();
+const MANIFEST = 'scripts/surfaces/_schema/converted-surfaces.json';
+const SURFACE_RE = /^scripts\/surfaces\/.+\/surfaces\/[^/]+\.descriptor\.json$/;
+const SAMPLE = 'scripts/surfaces/parcel_product/F01/surfaces/mobile_parcel_search.descriptor.json';
+const BAD_ERRORS = new Set(['none', 'UNRESEARCHED']);
+
+export interface Descriptor {
+  kind?: string;
+  identity?: { id?: string };
+  programme?: { scope?: string };
+  inputs?: { answer?: string };
+  states?: Array<{ name?: string }>;
+  errors?: { answer?: string };
+}
+export interface Scanned { path: string; d: Descriptor }
+
+const readJson = <T>(rel: string): T => JSON.parse(readFileSync(join(ROOT, rel), 'utf8')) as T;
+const scope = readJson<{ scope: { programme_scopes: string[]; expected_count: number } }>(MANIFEST).scope;
+
+/** Problems of the in-scope SURFACE descriptors that read a contract. Pure: descriptors and scopes in, one line per problem out. */
+export function failureStateProblems(list: Scanned[], scopes: readonly string[]): string[] {
+  const out: string[] = [];
+  for (const { path, d } of list) {
+    if (d.kind !== 'SURFACE' || !scopes.includes(d.programme?.scope ?? '')) continue;
+    if (d.inputs?.answer !== 'contract') continue;
+    const id = d.identity?.id ?? path;
+    if (!(d.states ?? []).some((s) => s.name === 'error')) out.push(`${id}: reads a contract but declares no error state`);
+    const ea = d.errors?.answer ?? 'none';
+    if (BAD_ERRORS.has(ea)) out.push(`${id}: reads a contract but its errors answer is ${ea}`);
+  }
+  return out;
+}
+
+function scanned(): Scanned[] {
+  return execFileSync('git', ['ls-files', '--', 'scripts/surfaces/'], { cwd: ROOT, stdio: 'pipe' })
+    .toString()
+    .split('\n')
+    .map((s) => s.trim())
+    .filter((f) => SURFACE_RE.test(f))
+    .map((f) => ({ path: f, d: readJson<Descriptor>(f) }));
+}
+
+const copy = (mut: (d: Descriptor) => void): Scanned => {
+  const d = structuredClone(readJson<Descriptor>(SAMPLE));
+  mut(d);
+  return { path: `${SAMPLE} (copy)`, d };
+};
+const noErrorState = (d: Descriptor) => { d.states = (d.states ?? []).filter((s) => s.name !== 'error'); };
+

+    const inScope = list.filter(({ d }) => d.kind === 'SURFACE' && scope.programme_scopes.includes(d.programme?.scope ?? ''));
+    const readers = inScope.filter(({ d }) => d.inputs?.answer === 'contract');
+    console.log(`fe.54: scanned ${list.length} surface descriptors, ${inScope.length} in scope, ${readers.length} read a contract, ${problems.length} problems`);
+    expect(problems, `fe.54 (Spec 128 Rule 12, 128:39) contract readers with no declared failure state:\n  ${problems.join('\n  ')}`).toEqual([]);
+    expect(inScope.length).toBe(scope.expected_count);
+  });
+
+  it('B. RED planted (KNOWN_BAD, in memory): each defect in a copy of mobile_parcel_search gives exactly one problem', () => {
+    const cases: Array<[Scanned, string]> = [
+      [copy(noErrorState), 'mobile_parcel_search: reads a contract but declares no error state'],
+      [copy((d) => { d.errors = { ...d.errors, answer: 'none' }; }), 'mobile_parcel_search: reads a contract but its errors answer is none'],
+      [copy((d) => { d.errors = { ...d.errors, answer: 'UNRESEARCHED' }; }), 'mobile_parcel_search: reads a contract but its errors answer is UNRESEARCHED'],
+    ];
+    for (const [s, line] of cases) expect(failureStateProblems([s], scope.programme_scopes)).toEqual([line]);
+  });
+
+  it('C. controls: the unchanged copy, a STATIC page with no contract input, and an out-of-scope descriptor give 0', () => {
+    expect(failureStateProblems([copy(() => {})], scope.programme_scopes)).toEqual([]);
+    const staticPage = copy((d) => { d.inputs = { answer: 'none' }; noErrorState(d); d.errors = { answer: 'none' }; });
+    expect(failureStateProblems([staticPage], scope.programme_scopes)).toEqual([]);
+    const outOfScope = copy((d) => { d.programme = { scope: 'estate_other' }; noErrorState(d); });
+    expect(failureStateProblems([outOfScope], scope.programme_scopes)).toEqual([]);
+  });
