+const ROOT = process.cwd();
+const NAMES = new Set(['alert', 'confirm', 'prompt']);
+const GLOBALS = new Set(['window', 'globalThis', 'self']);
+
+export interface Hit { file: string; line: number; what: string }
+
+/** Every call of alert, confirm or prompt, bare or on window, globalThis or self. Pure: path and text in, hits out. */
+export function findBlockingDialogs(rel: string, text: string): Hit[] {
+  const kind = rel.endsWith('.tsx') ? ts.ScriptKind.TSX : ts.ScriptKind.TS;
+  const sf = ts.createSourceFile(rel, text, ts.ScriptTarget.Latest, true, kind);
+  const hits: Hit[] = [];
+  const visit = (n: ts.Node): void => {
+    if (ts.isCallExpression(n)) {
+      const c = n.expression;
+      let what: string | null = null;
+      if (ts.isIdentifier(c) && NAMES.has(c.text)) what = `${c.text}()`;
+      else if (ts.isPropertyAccessExpression(c) && ts.isIdentifier(c.expression) && GLOBALS.has(c.expression.text) && NAMES.has(c.name.text)) {
+        what = `${c.expression.text}.${c.name.text}()`;
+      }
+      if (what) hits.push({ file: rel, line: sf.getLineAndCharacterOfPosition(n.getStart(sf)).line + 1, what });
+    }
+    ts.forEachChild(n, visit);
+  };
+  visit(sf);
+  return hits;
+}

+}

+export interface Hit { file: string; line: number; what: string }

+/** Every non-null assertion with no comment on its own line or the line above. Pure: path and text in, hits out. */
+export function findUndocumentedBangs(rel: string, text: string): Hit[] {
+  const kind = rel.endsWith('.tsx') ? ts.ScriptKind.TSX : ts.ScriptKind.TS;
+  const sf = ts.createSourceFile(rel, text, ts.ScriptTarget.Latest, true, kind);
+  const lineOf = (pos: number): number => sf.getLineAndCharacterOfPosition(pos).line;
+  const commentLines = new Set<number>();
+  const seen = new Set<number>();
+  const addRanges = (ranges: ts.CommentRange[] | undefined): void => {
+    for (const r of ranges ?? []) {
+      if (seen.has(r.pos)) continue;
+      seen.add(r.pos);
+      for (let l = lineOf(r.pos); l <= lineOf(r.end); l++) commentLines.add(l);
+    }
+  };
+  const bangLines: number[] = [];
+  const visit = (n: ts.Node): void => {
+    // JSX text is not trivia: `// x` inside it is text, not a comment.
+    if (n.kind !== ts.SyntaxKind.JsxText) {
+      addRanges(ts.getLeadingCommentRanges(text, n.pos));
+      addRanges(ts.getTrailingCommentRanges(text, n.end));
+    }
+    if (ts.isNonNullExpression(n)) bangLines.push(lineOf(n.end - 1));
+    for (const c of n.getChildren(sf)) visit(c);
+  };
+  visit(sf);
+  return bangLines
+    .filter((l) => !commentLines.has(l) && !commentLines.has(l - 1))
+    .map((l) => ({ file: rel, line: l + 1, what: 'undocumented !' }));
+}

+export interface PinProblem { file: string; count: number; ceiling: number | null }
+
+/** The ceiling refuses growth only: a file not in the pin with hits, or a count above its pin. A count below, a file at 0 or a gone file is no problem. */
+export function pinProblems(actual: Readonly<Record<string, number>>, pin: Readonly<Record<string, number>>): PinProblem[] {
+  const out: PinProblem[] = [];
+  for (const [file, count] of Object.entries(actual)) {
+    if (count <= 0) continue;
+    const ceiling = pin[file];
+    if (ceiling === undefined) out.push({ file, count, ceiling: null });
+    else if (count > ceiling) out.push({ file, count, ceiling });
+  }
+  return out;
+}

+/** Every data fetch or global context read in one file. Pure: path and text in, hits out. */
+export function findUiImpurity(rel: string, text: string): Hit[] {
+  const kind = rel.endsWith('.tsx') ? ts.ScriptKind.TSX : ts.ScriptKind.TS;
+  const sf = ts.createSourceFile(rel, text, ts.ScriptTarget.Latest, true, kind);
+  const hits: Hit[] = [];
+  const at = (n: ts.Node): number => sf.getLineAndCharacterOfPosition(n.getStart(sf)).line + 1;
+  const visit = (n: ts.Node): void => {
+    if (ts.isCallExpression(n)) {
+      const c = n.expression;
+      if (ts.isIdentifier(c) && CALLS.has(c.text)) hits.push({ file: rel, line: at(n), what: `${c.text}()` });
+      else if (ts.isPropertyAccessExpression(c) && ts.isIdentifier(c.expression)) {
+        const name = `${c.expression.text}.${c.name.text}`;
+        if (MEMBER_CALLS.has(name)) hits.push({ file: rel, line: at(n), what: `${name}()` });
+      }
+    } else if (ts.isImportDeclaration(n) && ts.isStringLiteral(n.moduleSpecifier) && n.moduleSpecifier.text === QUERY_PACKAGE) {
+      hits.push({ file: rel, line: at(n), what: `import from '${QUERY_PACKAGE}'` });
+    }
+    ts.forEachChild(n, visit);
+  };
+  visit(sf);
+  return hits;
+}
